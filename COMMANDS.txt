## Lorenz
cd examples/lorenz
python3 generate_lorenz.py --delay_dim 10 --delay_steps 3

cd ..
python3 show_input.py --data lorenz/delay_xcoordinate_d10_3.npz --n_ics 20

python3 train.py --data lorenz/delay_xcoordinate_d10_3.npz \
  --latent_dim 3 \
  --poly_order 2 \
  --widths 64,32 \
  --activation tanh \
  --loss_weight_decoder 1.0 \
  --loss_weight_sindy_x 1e-4 \
  --loss_weight_sindy_z 1e-3 \
  --loss_weight_coord 0.2 \
  --loss_weight_cons 0.04 \
  --loss_weight_sindy_reg 1e-3 \
  --coefficient_threshold 0.15 \
  --threshold_frequency 500 \
  --max_epochs 2001 \
  --refinement_epochs 201 \
  --batch_size 1024 \
  --learning_rate 1e-3 \
  --patience 8 \
  --scheduler_type plateau \
  --scheduler_patience 500 \
  --scheduler_factor 0.5 \
  --scheduler_min_lr 1e-5 \
  --print_frequency 100 \
  --seed 42
